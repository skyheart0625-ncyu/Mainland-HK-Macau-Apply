function toggleRankSelect() {

    const identity = document.getElementById('identity').value;

    const rankGroup = document.getElementById('rankGroup');

    const meetGroup = document.getElementById('meetGroup');

   

    if (identity === 'admin_teacher') {

        rankGroup.style.display = 'block';

    } else {

        rankGroup.style.display = 'none';

    }

   

    if (identity === 'admin_teacher' || identity === 'contract_staff') {

        meetGroup.style.display = 'block';

    } else {

        meetGroup.style.display = 'none';

    }

}

 

function calculateGuidance() {

    const identity = document.getElementById('identity').value;

    const rank = document.getElementById('rank').value;

    const exitDateVal = document.getElementById('exitDate').value;

    const destination = document.getElementById('destination').value;

    const purpose = document.getElementById('purpose').value;

    const meetTarget = document.getElementById('meetTarget').value;

   

    const resultDiv = document.getElementById('result');

    const identityTag = document.getElementById('identityTag');

    const deadlineBadge = document.getElementById('deadlineBadge');

    const stepList = document.getElementById('stepList');

    const downloadList = document.getElementById('downloadList');

    const warningBox = document.getElementById('warningBox');

    const penaltyBox = document.getElementById('penaltyBox');

 

    if (!identity) {

        alert('請選擇身分類別！');

        return;

    }

 

    resultDiv.style.display = 'block';

    stepList.innerHTML = '';

    downloadList.innerHTML = '';

    deadlineBadge.style.display = 'none';

   

    let includesMainland = (destination === 'mainland' || destination === 'both');

    let includesHKMacau = (destination === 'hk_macau' || destination === 'both');

    let isTeacher = (identity === 'teacher');

   

    let identityText = document.getElementById('identity').options[document.getElementById('identity').selectedIndex].text;

    identityTag.innerText = identityText;

 

    // 計算最晚送件日

    if (exitDateVal && !isTeacher) {

        let exitDate = new Date(exitDateVal);

        let weeksToSubtract = 2;

 

        if (includesHKMacau) {

            if (purpose === 'official' && rank === 'rank12') weeksToSubtract = 3;

            if (meetTarget === 'yes') {

                if (rank === 'rank12') weeksToSubtract = 5;

                else weeksToSubtract = 2;

            }

        }

        if (includesMainland) weeksToSubtract = 2;

 

        let deadlineDate = new Date(exitDate);

        deadlineDate.setDate(deadlineDate.getDate() - (weeksToSubtract * 7));

       

        let yyyy = deadlineDate.getFullYear();

        let mm = String(deadlineDate.getMonth() + 1).padStart(2, '0');

        let dd = String(deadlineDate.getDate()).padStart(2, '0');

       

        deadlineBadge.innerText = `最晚須於 ${yyyy}-${mm}-${dd} 前（出境日 ${weeksToSubtract} 週前）完成送件！`;

        deadlineBadge.style.display = 'inline-block';

    }

 

    let steps = [];

    let warnings = [];

    let penalties = [];

    let files = [];

 

    // 固定連結網址

    const urlPortal = "https://www.ncyu.edu.tw/personnel/Subject/Detail/232082?nodeId=35206";

    const urlMAC = "https://www.mac.gov.tw/cp.aspx?n=015A70099E11C8A8";

 

    // --- 一、一般教師 ---

    if (isTeacher) {

        steps.push("<strong>【校內請假】</strong>循本校出國申請流程辦理並經核准。");

        steps.push("<strong>【行前動態登錄】</strong>赴陸港澳前至陸委會「國人赴陸港澳動態登錄系統」進行線上登錄。");

        steps.push("<strong>【完成手續】</strong>未兼任行政職教師無須送內政部或校長事前核准，亦無須填寫返臺通報表。");

       

        files.push(`🌐 <a href="${urlMAC}" target="_blank">大陸委員會—國人赴陸港澳動態登錄系統 (線上登錄)</a>`);

    } else {

        // --- 二、兼任行政職教師 / 公務人員 / 契約人員 ---

       

        // 1. 赴大陸程序與表件

        if (includesMainland) {

            steps.push("<strong><big>【赴大陸地區作業程序】</big></strong>");

            steps.push("循本校出國申請流程辦理並經核准。[cite: 3]");

           

            if (identity === 'admin_teacher' && (rank === 'rank11' || rank === 'rank12')) {

                steps.push(" <strong>至遲出境日 2 週前</strong>填具「簡任第 11 職等以上赴大陸地區申請表」，送人事室函報<strong>內政部核准</strong>。");

                steps.push("報經內政部許可始得赴陸。[cite: 3]");

                steps.push("赴陸前至陸委會「國人赴陸港澳動態登錄系統」進行線上登錄。");

                steps.push(" <strong>返臺 1 星期內</strong>填寫「赴陸人員返臺通報表」，送人事室<strong>通報內政部</strong>。");

               

                files.push(`📄 <a href="${urlPortal}" target="_blank">簡任第 11 職等以上人員進入大陸地區申請表 (下載)</a>`);

                files.push(`📄 <a href="${urlPortal}" target="_blank">赴陸人員返臺通報表 (下載)</a>`);

            } else {

                steps.push(" <strong>至遲出境日 2 週前</strong>填具「簡任第 10 職等以下赴大陸地區申請表」，送人事室辦理（會辦政風）。");

                steps.push("奉<strong>校長核准同意</strong>。");

                steps.push("赴陸前至陸委會「國人赴陸港澳動態登錄系統」進行線上登錄。");

                steps.push("5. <strong>返臺 1 星期內</strong>填寫「赴陸人員返臺通報表」，送人事室<strong>備查</strong>。");

               

                files.push(`📄 <a href="${urlPortal}" target="_blank">簡任第 10 職等以下人員赴大陸地區申請表 (下載)</a>`);

                files.push(`📄 <a href="${urlPortal}" target="_blank">赴陸人員返臺通報表 (下載)</a>`);

            }

            penalties.push("<strong>赴陸違規懲處：</strong>未依規定完成申請程序者，依公務人員考績法及學校相關法規懲處。");

        }

 

        // 2. 赴港澳程序與表件

        if (includesHKMacau) {

            steps.push("<strong><big>【赴港澳地區作業程序】</big></strong>");

            steps.push("1. 依校內出國申請流程辦理並經核准。");

 

            if (purpose === 'official') {

                if (rank === 'rank12') {

                    steps.push(" <strong>出境日 3 週前</strong>填具因公赴港澳通報表，送人事室通報<strong>陸委會</strong>。");

                } else {

                    steps.push(" <strong>出境日 2 週前</strong>填具因公赴港澳通報表，送人事室通報<strong>陸委會</strong>。");

                }

                files.push(`📄 <a href="${urlPortal}" target="_blank">公務人員因公赴香港澳門通報表 (下載)</a>`);

            } else {

                if (meetTarget === 'no') {

                    steps.push(" <strong>出境日 2 週前</strong>填具非因公赴港澳通報表送人事室。");

                    files.push(`📄 <a href="${urlPortal}" target="_blank">非因公務事由赴香港或澳門通報表 (下載)</a>`);

                }

            }

 

            if (meetTarget === 'yes') {

                if (rank === 'rank12') {

                    steps.push("3. <strong>出境日 5 週前</strong>填具會見或聯繫特定身分人員通報表送人事室通報。");

                } else {

                    steps.push("3. <strong>出境日 2 週前</strong>填具會見或聯繫特定身分人員通報表送人事室通報。");

                }

                files.push(`📄 <a href="${urlPortal}" target="_blank">赴香港或澳門會見或聯繫特定身分人員通報表 (下載)</a>`);

            }

 

            steps.push("至陸委會官網「國人赴陸港澳動態登錄系統」進行登錄，並將<strong>登錄結果影送人事室留存</strong>。");

            steps.push("5. <strong>臨時會見補通報：</strong>若於港澳期間臨時會見或聯繫特定身分人員未經事前通報，應於<strong>返臺 3 日內</strong>主動填具通報表送人事室通報陸委會。");

 

            penalties.push("<strong>赴港澳未依規通報懲處：</strong>依校內人事法規懲處。");

        }

 

        files.push(`🌐 <a href="${urlMAC}" target="_blank">大陸委員會—國人赴陸港澳動態登錄系統 (線上登錄)</a>`);

    }

 

    // --- 通用注意事項 ---

    warnings.push("<strong>⚠️ 國立嘉義大學重點宣導與提醒：</strong>");

    warnings.push(" <strong>出境日期以「查驗通關日」為準：</strong>若出關與搭機時間不同（例如凌晨 01:00 飛機，在前一日晚上 23:00 查驗出關），<strong>申請赴陸港澳之起始日期應以「前一日（出關日）」為準</strong>，避免因許可生效日期不符而無法出境。");

    warnings.push(" <strong>過境轉機：</strong>搭乘航班經中國大陸或港澳地區機場「入境轉機」或「不入境之過境轉機」，均須依上述規定辦理。");

    warnings.push(" <strong>不分平日假日：</strong>本校教師不分平日、假日赴港澳/大陸，行前均應至線上簽核差勤系統完成出國申請並經核准。");

    warnings.push(" <strong>人事室承辦窗口：</strong>人事室承辦人陳小姐（聯絡電話：05-2717196）。");

 

    // 渲染至 HTML

    steps.forEach(step => {

        let li = document.createElement('li');

        li.innerHTML = step;

        stepList.appendChild(li);

    });

 

    files.forEach(file => {

        let li = document.createElement('li');

        li.innerHTML = file;

        downloadList.appendChild(li);

    });

 

    warningBox.innerHTML = warnings.join('<br>');

    penaltyBox.innerHTML = penalties.length > 0 ? penalties.join('<br>') : "<strong>懲處規定：</strong>一般教師依校內一般請假差勤規定辦理。";

}
